# Run Index Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a local, queryable index over past monerosim simulation runs, with a Python CLI that is fully useful before any LLM is involved.

**Architecture:** A new `scripts/runs/` package walks configurable source roots, extracts one normalized JSON record per run directory (reading the 0.1 MB `summary.txt`, never the 43 MB `final_report.json`), and writes them to `~/.monerosim/run_index.jsonl`. Query subcommands filter that file. The canonical `summary.txt` parser is moved into this package and the two existing consumers re-point at it, so the repo ends with one parser rather than three.

**Tech Stack:** Python 3 (stdlib + PyYAML, both already used by the repo), pytest (configured in `pyproject.toml`, `testpaths = ["agents","scripts","tests"]`).

**Spec:** `docs/superpowers/specs/2026-09-20-run-index-design.md`

## Global Constraints

- Scope is **increments 1 and 2 only** (extractor + CLI, then `scan`). The `run_sim.sh` run-time capture and the model layer are explicitly deferred.
- Index path: `~/.monerosim/run_index.jsonl`. Local, never committed.
- Read `summary.txt` for outcome fields. Use `monitoring/final_report.json` **only** when `summary.txt` is absent.
- **Never** read completion from `final_report.json`'s `status` field — it reports `running` on runs that finished cleanly.
- Criteria are tri-state: `true` / `false` / `"n/a"`. **Never treat `"n/a"` as a pass.** `"n/a"` is a truthy string, so `all(values)` over an all-skipped set wrongly returns `True`.
- **No single pass/fail field** in the record. Store `exit_code`, the raw criteria dict, and a derived `tx_workload` flag; let queries compose.
- Strictly **read-only** over run artifacts. `scan` never writes to the volume it walks.
- When walking a run directory, **skip `shadow.data/`** — it holds hundreds of GB of LMDB.
- Default cutoff: runs with a `YYYYMMDD_` prefix earlier than `20260513` (v0.1.0) are skipped. Configurable, never hardcoded at a call site.
- Do **not** add a third parser of `summary.txt`. One canonical implementation, imported by all consumers.
- Run tests with `venv/bin/python -m pytest`.
- Every record carries `schema_version`. A bump invalidates the index and forces a full rebuild; records are never migrated in place.

# scripts/runs/config.py
"""Source roots and the version cutoff for the run index."""
from __future__ import annotations

from dataclasses import dataclass, field
from pathlib import Path
from typing import List, Optional

import yaml

# v0.1.0 "Pancake Plethora", 2026-05-13 — the first public beta. Runs older than
# this predate the current run-directory layout and are skipped. Configurable:
# the 23 pre-cutoff runs on the backup volume do have real configs, they are just
# from a period not worth indexing today.
DEFAULT_CUTOFF = "20260513"

MONEROSIM_HOME = Path.home() / ".monerosim"
INDEX_PATH = MONEROSIM_HOME / "run_index.jsonl"
CONFIG_PATH = MONEROSIM_HOME / "runs.yaml"

REPO_ROOT = Path(__file__).resolve().parent.parent.parent


@dataclass
class Source:
    """One root to search for run directories. Order is precedence."""
    label: str
    root: Path


@dataclass
class RunsConfig:
    sources: List[Source] = field(default_factory=list)
    cutoff: str = DEFAULT_CUTOFF


def default_sources() -> List[Source]:
    return [
        Source("live", REPO_ROOT / "archived_runs"),
        Source("curated", Path.home() / "basement_monerosim"),
        Source("cold", Path("/mnt/remote_spinny/monerosim_backups")),
        Source("metrics", REPO_ROOT / "analysis" / "eclipse" / "results"),
    ]


def load_config(path: Optional[Path] = None) -> RunsConfig:
    """Load config, falling back to defaults for anything unspecified."""
    path = Path(path) if path is not None else CONFIG_PATH
    if not path.exists():
        return RunsConfig(sources=default_sources(), cutoff=DEFAULT_CUTOFF)
    raw = yaml.safe_load(path.read_text()) or {}
    sources = [
        Source(str(s["label"]), Path(str(s["root"])))
        for s in raw.get("sources", [])
    ] or default_sources()
    return RunsConfig(sources=sources, cutoff=str(raw.get("cutoff", DEFAULT_CUTOFF)))
```

# scripts/runs/summary.py
"""Canonical parser for a run's summary.txt.

This is the ONE implementation. scripts/append_run_history.py and
scripts/smoke_assertions.py import from here; do not copy it again.

Criteria are tri-state since 2026-09-20: True / False / "n/a". "n/a" is a truthy
string, so a bare all(values) counts a skipped criterion as a pass. Always use
all_success_criteria_pass, which filters first.
"""
from __future__ import annotations

import re
from pathlib import Path
from typing import Any, Dict, Optional

NA_VALUES = ("n/a", "na", "not_applicable")

# Label -> record key. "Blocks propagated" is legacy (pre-2026-09-20) and meant
# funded nodes, not propagation; it is kept for reading old runs only.
CRITERIA_LABELS = {
    "Blocks created": "blocks_created",
    "Nodes funded": "nodes_funded",
    "Blocks propagated (actual)": "actual_blocks_propagated",
    "Blocks propagated": "blocks_propagated",
    "Transactions broadcast": "transactions_created_broadcast",
    "Transactions in blocks": "transactions_in_blocks",
}


def parse_wall_time(s: str) -> Optional[int]:
    """"5h 38m" -> seconds. Returns None if nothing parses."""
    if not s:
        return None
    total, found = 0, False
    for value, unit in re.findall(r"(\d+)\s*([hms])", s):
        found = True
        total += int(value) * {"h": 3600, "m": 60, "s": 1}[unit]
    return total if found else None


def _int(text: str, pattern: str) -> Optional[int]:
    m = re.search(pattern, text)
    return int(m.group(1)) if m else None


def parse_summary(summary_path: Path) -> Dict[str, Any]:
    """Parse summary.txt. Missing fields come back as None, never raise."""
    text = Path(summary_path).read_text(errors="replace")

    criteria: Dict[str, Any] = {}
    for label, key in CRITERIA_LABELS.items():
        m = re.search(rf"^\s+{re.escape(label)}\s+(PASS|FAIL|N/A)\s*$",
                      text, re.MULTILINE)
        if m:
            criteria[key] = "n/a" if m.group(1) == "N/A" else (m.group(1) == "PASS")

    applicable = [v for v in criteria.values() if not isinstance(v, str)]

    m = re.search(r"Wall time:\s*(.+)", text)
    return {
        "wall_time_s": parse_wall_time(m.group(1).strip()) if m else None,
        "exit_code": _int(text, r"Exit code:\s*(\d+)"),
        "success_criteria": criteria,
        "all_success_criteria_pass": bool(applicable) and all(applicable),
        "nodes_online": _int(text, r"Nodes online:\s*(\d+)"),
        "sync_pct": _int(text, r"Sync:\s*(\d+)"),
        "block_height": _int(text, r"Block height:\s*(\d+)"),
        "blocks_mined": _int(text, r"Blocks mined:\s*(\d+)"),
        "alerts": _int(text, r"Alerts:\s*(\d+)"),
        "tx_created": _int(text, r"Created:\s*(\d+)"),
        "tx_in_blocks": _int(text, r"In blocks:\s*(\d+)"),
    }
```

# scripts/runs/discover.py
"""Find run directories under configured roots."""
from __future__ import annotations

import re
from pathlib import Path
from typing import List, Tuple

from .config import Source

RUN_ID_RE = re.compile(r"^\d{8}_\d{6}_")

# Any one of these marks a directory as a run. Checked cheaply, no recursion.
RUN_MARKERS = (
    "summary.txt",
    "input_config.yaml",
    "shadow_run.log",
    "eclipse_metrics.jsonl",
    "monitoring/final_report.json",
)

# Never descend into these: shadow.data holds hundreds of GB of LMDB.
SKIP_DIRS = {"shadow.data", "daemon_logs", "shadow_output", ".git", "venv"}


def is_run_dir(path: Path) -> bool:
    if not path.is_dir():
        return False
    return any((path / m).exists() for m in RUN_MARKERS)


def passes_cutoff(run_id: str, cutoff: str) -> bool:
    return run_id[:8] >= cutoff


def find_runs(sources: List[Source], cutoff: str) -> List[Tuple[str, Path, str]]:
    """Return (run_id, path, source_label). Earlier sources win on duplicates."""
    seen: dict = {}
    for source in sources:
        root = Path(source.root)
        if not root.is_dir():
            continue  # a cold volume may simply not be mounted
        try:
            entries = sorted(root.iterdir())
        except OSError:
            continue
        for entry in entries:
            name = entry.name
            if not RUN_ID_RE.match(name) or not passes_cutoff(name, cutoff):
                continue
            if name in seen or not is_run_dir(entry):
                continue
            seen[name] = (name, entry, source.label)
    return [seen[k] for k in sorted(seen)]
```

# scripts/runs/extract.py
"""Build one index record from one run directory. Read-only."""
from __future__ import annotations

import csv
import gzip
import hashlib
import json
import re
from pathlib import Path
from typing import Any, Dict, Optional

import yaml

from .summary import parse_summary

SCHEMA_VERSION = 1

# "Progress: 100% — simulated: 05:58:15.355/06:00:00, realtime: 04:22:00, processes failed: 0"
# Note the em dash (U+2014). Times are H:MM:SS with optional fractional seconds.
PROGRESS_RE = re.compile(
    r"Progress:\s*(?P<pct>\d+)%.*?simulated:\s*(?P<sim>[\d:.]+)/(?P<target>[\d:.]+),"
    r"\s*realtime:\s*(?P<real>[\d:]+),\s*processes failed:\s*(?P<failed>\d+)"
)


def _hms_to_seconds(s: str) -> Optional[int]:
    try:
        parts = [float(p) for p in s.split(":")]
    except ValueError:
        return None
    total = 0.0
    for p in parts:
        total = total * 60 + p
    return int(total)


def parse_shadow_progress(text: str) -> Dict[str, Any]:
    """Parse the LAST progress line plus the completion marker."""
    out: Dict[str, Any] = {
        "sim_achieved_s": None, "sim_target_s": None,
        "wall_clock_s": None, "processes_failed": None, "completed": False,
    }
    matches = list(PROGRESS_RE.finditer(text))
    if matches:
        m = matches[-1]
        out["sim_achieved_s"] = _hms_to_seconds(m.group("sim"))
        out["sim_target_s"] = _hms_to_seconds(m.group("target"))
        out["wall_clock_s"] = _hms_to_seconds(m.group("real"))
        out["processes_failed"] = int(m.group("failed"))
    out["completed"] = "Shadow completed successfully" in text
    return out


def _read_maybe_gz(path: Path, name: str) -> Optional[str]:
    """Read `name` or `name.gz` from a run dir. Basement-tier runs are gzipped."""
    plain = path / name
    if plain.exists():
        try:
            return plain.read_text(errors="replace")
        except OSError:
            return None
    gz = path / (name + ".gz")
    if gz.exists():
        try:
            with gzip.open(gz, "rt", errors="replace") as f:
                return f.read()
        except OSError:
            return None
    return None


def summary_from_final_report(path: Path) -> Optional[Dict[str, Any]]:
    """Fallback for the ~15 runs with no summary.txt.

    monitoring/final_report.json carries the same fields but is ~43 MB because it
    embeds the full historical_data series, so this is a fallback only, never the
    primary path. Its `status` field is NEVER used: it reports "running" on runs
    that finished cleanly.
    """
    report = path / "monitoring" / "final_report.json"
    if not report.exists():
        return None
    try:
        data = json.loads(report.read_text(errors="replace"))
    except (OSError, json.JSONDecodeError):
        return None
    summary = data.get("summary") or {}
    criteria = summary.get("success_criteria") or {}
    applicable = [v for v in criteria.values() if not isinstance(v, str)]
    return {
        "wall_time_s": None,
        "exit_code": None,
        "success_criteria": criteria,
        "all_success_criteria_pass": bool(applicable) and all(applicable),
        "nodes_online": summary.get("total_nodes"),
        "sync_pct": summary.get("avg_sync_percentage"),
        "block_height": summary.get("max_height"),
        "blocks_mined": summary.get("total_blocks_mined"),
        "alerts": summary.get("alert_count"),
        "tx_created": summary.get("total_transactions_created"),
        "tx_in_blocks": summary.get("total_transactions_in_blocks"),
    }


def peak_rss_mb(csv_path: Path) -> Optional[float]:
    try:
        with open(csv_path, newline="") as f:
            values = [
                float(row["total_rss_mb"])
                for row in csv.DictReader(f)
                if row.get("total_rss_mb")
            ]
    except (OSError, ValueError, KeyError):
        return None
    return max(values) if values else None


def leading_comment(config_path: Path) -> str:
    """The '#' block at the top of a config: the human-written intent of the run."""
    lines = []
    try:
        with open(config_path, errors="replace") as f:
            for line in f:
                if line.startswith("#"):
                    lines.append(line.lstrip("#").strip())
                elif line.strip() == "":
                    if lines:
                        break
                else:
                    break
    except OSError:
        return ""
    return "\n".join(lines).strip()


def _config_facts(config_path: Path) -> Dict[str, Any]:
    facts: Dict[str, Any] = {
        "stop_time": None, "agent_count": None, "gml_path": None,
        "mining_mode": None, "reachable_fraction": None,
        "node_implementations": None, "daemons": [],
    }
    try:
        data = yaml.safe_load(config_path.read_text(errors="replace")) or {}
    except Exception:
        return facts
    general = data.get("general") or {}
    facts["stop_time"] = general.get("stop_time")
    facts["reachable_fraction"] = general.get("reachable_fraction")
    facts["node_implementations"] = general.get("node_implementations")
    mining = general.get("mining") or {}
    facts["mining_mode"] = mining.get("mode")
    facts["gml_path"] = (data.get("network") or {}).get("path")
    agents = data.get("agents") or {}
    if isinstance(agents, dict):
        facts["agent_count"] = len(agents)
        facts["daemons"] = sorted({
            a.get("daemon") for a in agents.values()
            if isinstance(a, dict) and a.get("daemon")
        })
    return facts


def extract_record(run_id: str, path: Path, source_label: str) -> Dict[str, Any]:
    """One record. Fields that cannot be determined are None and named in `missing`."""
    path = Path(path)
    missing = []
    rec: Dict[str, Any] = {
        "schema_version": SCHEMA_VERSION,
        "run_id": run_id,
        "timestamp": run_id[:15],
        "run_name": run_id[16:] or None,
        "source_root": source_label,
        "path": str(path),
    }

    config_path = path / "input_config.yaml"
    if config_path.exists():
        raw = config_path.read_bytes()
        rec["config_sha256"] = hashlib.sha256(raw).hexdigest()
        rec["description"] = leading_comment(config_path)
        rec.update(_config_facts(config_path))
    else:
        rec["config_sha256"] = None
        rec["description"] = ""
        rec.update({k: None for k in
                    ("stop_time", "agent_count", "gml_path", "mining_mode",
                     "reachable_fraction", "node_implementations")})
        rec["daemons"] = []
        missing.append("config")

    shadow_text = _read_maybe_gz(path, "shadow_run.log")
    if shadow_text is not None:
        rec.update(parse_shadow_progress(shadow_text))
    else:
        rec.update({"sim_achieved_s": None, "sim_target_s": None,
                    "wall_clock_s": None, "processes_failed": None,
                    "completed": None})
        missing.append("wall_clock_s")

    summary_path = path / "summary.txt"
    s = parse_summary(summary_path) if summary_path.exists() else summary_from_final_report(path)
    if s is not None:
        rec["exit_code"] = s["exit_code"]
        rec["success_criteria"] = s["success_criteria"]
        rec["all_success_criteria_pass"] = s["all_success_criteria_pass"]
        rec["nodes_online"] = s["nodes_online"]
        rec["sync_pct"] = s["sync_pct"]
        rec["block_height"] = s["block_height"]
        rec["blocks_mined"] = s["blocks_mined"]
        rec["alerts"] = s["alerts"]
        rec["tx_created"] = s["tx_created"]
        rec["tx_in_blocks"] = s["tx_in_blocks"]
        # Derived: did this run have a transaction workload at all? Used to skip
        # transaction criteria that are meaningless for mining-only configs.
        rec["tx_workload"] = bool(s["tx_created"])
        if rec["wall_clock_s"] is None and s["wall_time_s"] is not None:
            rec["wall_clock_s"] = s["wall_time_s"]
            if "wall_clock_s" in missing:
                missing.remove("wall_clock_s")
        if rec["agent_count"] is None:
            rec["agent_count"] = s["nodes_online"]
    else:
        for key in ("exit_code", "nodes_online", "sync_pct", "block_height",
                    "blocks_mined", "alerts", "tx_created", "tx_in_blocks"):
            rec[key] = None
        rec["success_criteria"] = {}
        rec["all_success_criteria_pass"] = None
        rec["tx_workload"] = None
        missing.append("success_criteria")

    rec["peak_total_rss_mb"] = peak_rss_mb(path / "memory_samples.csv")
    if rec["peak_total_rss_mb"] is None:
        missing.append("peak_total_rss_mb")

    if rec.get("sim_achieved_s") and rec.get("wall_clock_s"):
        rec["sim_wall_ratio"] = round(rec["sim_achieved_s"] / rec["wall_clock_s"], 3)
    else:
        rec["sim_wall_ratio"] = None
    if rec.get("peak_total_rss_mb") and rec.get("agent_count"):
        rec["rss_per_agent_mb"] = round(
            rec["peak_total_rss_mb"] / rec["agent_count"], 2)
    else:
        rec["rss_per_agent_mb"] = None

    rec["missing"] = missing
    return rec
```

```python
# scripts/runs/store.py
"""Read and write the JSONL run index."""
from __future__ import annotations

import json
from pathlib import Path
from typing import Any, Dict, List

from .config import Source
from .discover import find_runs
from .extract import SCHEMA_VERSION, extract_record


def build_index(sources: List[Source], cutoff: str) -> List[Dict[str, Any]]:
    return [
        extract_record(run_id, path, label)
        for run_id, path, label in find_runs(sources, cutoff)
    ]


def write_index(records: List[Dict[str, Any]], path: Path) -> None:
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    with open(path, "w") as f:
        for rec in records:
            f.write(json.dumps(rec, sort_keys=True) + "\n")


def load_index(path: Path) -> List[Dict[str, Any]]:
    """Load records. A schema mismatch yields [] -- rebuild, never migrate."""
    path = Path(path)
    if not path.exists():
        return []
    records = []
    try:
        for line in path.read_text().splitlines():
            if not line.strip():
                continue
            rec = json.loads(line)
            if rec.get("schema_version") != SCHEMA_VERSION:
                return []
            records.append(rec)
    except (OSError, json.JSONDecodeError):
        return []
    return records


def staleness(records, sources: List[Source], cutoff: str) -> Dict[str, int]:
    """Runs present on disk but absent from the index, per source label."""
    indexed = {r["run_id"] for r in records}
    out: Dict[str, int] = {}
    for run_id, _path, label in find_runs(sources, cutoff):
        if run_id not in indexed:
            out[label] = out.get(label, 0) + 1
    return out
```

# scripts/runs/query.py
"""Filter predicates over index records."""
from __future__ import annotations

from typing import Any, Dict, List, Optional


def apply_filters(
    records: List[Dict[str, Any]],
    after: Optional[str] = None,
    before: Optional[str] = None,
    config_sha: Optional[str] = None,
    daemon: Optional[str] = None,
    completed: Optional[bool] = None,
    exit_code: Optional[int] = None,
    criteria_pass: Optional[bool] = None,
    min_agents: Optional[int] = None,
    max_failed_processes: Optional[int] = None,
) -> List[Dict[str, Any]]:
    out = []
    for r in records:
        if after and r.get("run_id", "")[:8] < after:
            continue
        if before and r.get("run_id", "")[:8] > before:
            continue
        if config_sha and not str(r.get("config_sha256") or "").startswith(config_sha):
            continue
        if daemon and daemon not in (r.get("daemons") or []):
            continue
        if completed is not None and r.get("completed") is not completed:
            continue
        if exit_code is not None and r.get("exit_code") != exit_code:
            continue
        if criteria_pass is not None and r.get("all_success_criteria_pass") is not criteria_pass:
            continue
        if min_agents is not None and (r.get("agent_count") or 0) < min_agents:
            continue
        if max_failed_processes is not None:
            failed = r.get("processes_failed")
            if failed is None or failed > max_failed_processes:
                continue
        out.append(r)
    return out


def search(records: List[Dict[str, Any]], text: str) -> List[Dict[str, Any]]:
    needle = (text or "").lower()
    fields = ("run_id", "run_name", "description")
    return [
        r for r in records
        if any(needle in str(r.get(f) or "").lower() for f in fields)
    ]
```

# scripts/runs/__main__.py
"""CLI for the monerosim run index.

    python -m scripts.runs index
    python -m scripts.runs ls --min-agents 1000 --completed
    python -m scripts.runs show 20260918_175657_p4_gamma_relay
    python -m scripts.runs find eclipse --json
    python -m scripts.runs scan /mnt/remote_spinny
"""
from __future__ import annotations

import argparse
import json
import sys
from pathlib import Path

from .config import INDEX_PATH, load_config
from .query import apply_filters, search
from .store import build_index, load_index, staleness, write_index


def _load_or_warn(cfg, args):
    records = load_index(args.index)
    if not records:
        print("index is empty or was invalidated by a schema change; "
              "run 'python -m scripts.runs index'", file=sys.stderr)
        return records
    behind = staleness(records, cfg.sources, cfg.cutoff)
    for label, n in behind.items():
        print(f"index is {n} runs behind {label} - "
              f"run 'python -m scripts.runs index'", file=sys.stderr)
    return records


def _emit(records, as_json: bool) -> None:
    if as_json:
        print(json.dumps(records, indent=2, sort_keys=True))
        return
    if not records:
        print("no matching runs")
        return
    print(f"{'RUN ID':42s} {'AGENTS':>7s} {'EXIT':>5s} {'WALL':>9s}  NAME")
    for r in records:
        wall = r.get("wall_clock_s")
        wall_s = f"{wall // 3600}h{(wall % 3600) // 60:02d}m" if wall else "-"
        print(f"{r.get('run_id',''):42s} {str(r.get('agent_count') or '-'):>7s} "
              f"{str(r.get('exit_code') if r.get('exit_code') is not None else '-'):>5s} "
              f"{wall_s:>9s}  {r.get('run_name') or ''}")


def main(argv=None) -> int:
    parser = argparse.ArgumentParser(prog="python -m scripts.runs")
    parser.add_argument("--index", type=Path, default=INDEX_PATH)
    parser.add_argument("--config", type=Path, default=None)
    sub = parser.add_subparsers(dest="cmd", required=True)

    sub.add_parser("index", help="build or refresh the local index")

    for name, helptext in (("ls", "filter and list runs"),
                           ("find", "search descriptions and names")):
        p = sub.add_parser(name, help=helptext)
        if name == "find":
            p.add_argument("text")
        p.add_argument("--after")
        p.add_argument("--before")
        p.add_argument("--config-sha")
        p.add_argument("--daemon")
        p.add_argument("--min-agents", type=int)
        p.add_argument("--exit-code", type=int)
        p.add_argument("--completed", action="store_true", default=None)
        p.add_argument("--criteria-pass", action="store_true", default=None)
        p.add_argument("--max-failed-processes", type=int)
        p.add_argument("--json", action="store_true")

    p = sub.add_parser("show", help="print one full record")
    p.add_argument("run_id")
    p.add_argument("--json", action="store_true", default=True)

    p = sub.add_parser("scan", help="discover run data under a path")
    p.add_argument("path", type=Path)
    p.add_argument("--json", action="store_true")

    args = parser.parse_args(argv)
    cfg = load_config(args.config)

    if args.cmd == "index":
        records = build_index(cfg.sources, cfg.cutoff)
        write_index(records, args.index)
        print(f"indexed {len(records)} runs -> {args.index}")
        return 0

    if args.cmd == "scan":
        from .discover import scan_path
        result = scan_path(args.path, cfg.cutoff)
        print(json.dumps(result, indent=2) if args.json else _format_scan(result))
        return 0

    records = _load_or_warn(cfg, args)

    if args.cmd == "show":
        for r in records:
            if r.get("run_id") == args.run_id:
                print(json.dumps(r, indent=2, sort_keys=True))
                return 0
        print(f"no run with id {args.run_id}", file=sys.stderr)
        return 1

    if args.cmd == "find":
        records = search(records, args.text)

    records = apply_filters(
        records, after=args.after, before=args.before,
        config_sha=args.config_sha, daemon=args.daemon,
        completed=args.completed, exit_code=args.exit_code,
        criteria_pass=args.criteria_pass, min_agents=args.min_agents,
        max_failed_processes=args.max_failed_processes,
    )
    _emit(records, args.json)
    return 0


def _format_scan(result) -> str:
    lines = [f"found {result['run_count']} run directories under {result['path']}"]
    if result["run_count"]:
        lines.append(f"  date range: {result['earliest']} .. {result['latest']}")
        lines.append(f"  after cutoff: {result['after_cutoff']}")
        lines.append("")
        lines.append("add to ~/.monerosim/runs.yaml:")
        lines.append("sources:")
        lines.append(f"  - label: <name>\n    root: {result['path']}")
    return "\n".join(lines)


if __name__ == "__main__":
    sys.exit(main())
```

# append to scripts/runs/discover.py
from typing import Dict


def scan_path(path: Path, cutoff: str, max_depth: int = 3) -> Dict[str, object]:
    """Walk a path looking for run directories. Read-only, depth-limited.

    Never descends into SKIP_DIRS: shadow.data alone can hold hundreds of GB.
    """
    root = Path(path)
    found = []

    def walk(current: Path, depth: int) -> None:
        if depth > max_depth:
            return
        try:
            entries = sorted(current.iterdir())
        except OSError:
            return
        for entry in entries:
            if not entry.is_dir() or entry.name in SKIP_DIRS:
                continue
            if RUN_ID_RE.match(entry.name) and is_run_dir(entry):
                found.append(entry.name)
                continue  # a run dir is a leaf for scanning purposes
            walk(entry, depth + 1)

    walk(root, 0)
    ids = sorted(found)
    return {
        "path": str(root),
        "run_count": len(ids),
        "after_cutoff": sum(1 for i in ids if passes_cutoff(i, cutoff)),
        "earliest": ids[0][:8] if ids else None,
        "latest": ids[-1][:8] if ids else None,
        "sample": ids[:5],
    }
